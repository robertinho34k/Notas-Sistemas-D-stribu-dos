quando estende a classe thread - memoria compartilhada ou nao
    com memoria compartilhada - synchronize


EXEMPLO JAVA:
    class MinhaThread extends Thread {
    public void run() {
        System.out.println("Thread executando!");
    }
}

public class Main {
    public static void main(String[] args) {
        MinhaThread t = new MinhaThread();
        t.start();  // inicia a thread
    }
}


EXEMPLO PYTHON:
import threading

def tarefa():
    print("Thread executando!")

t = threading.Thread(target=tarefa)
t.start()


EXEPLO C#:
using System;
using System.Threading;

class Program {
    static void MinhaThread() {
        Console.WriteLine("Thread executando!");
    }

    static void Main() {
        Thread t = new Thread(new ThreadStart(MinhaThread));
        t.Start();
    }
}
